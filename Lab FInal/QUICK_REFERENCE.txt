┌──────────────────────────────────────────────────────────────────────────────┐
│ 📊 MONGODB QUERIES                                                           │
└──────────────────────────────────────────────────────────────────────────────┘

  Total Revenue:
  Order.aggregate([{ $group: { _id: null, total: { $sum: "$totalAmount" }}}])

  Top Product:
  Order.aggregate([
    { $unwind: "$items" },
    { $group: { _id: "$items.product", qty: { $sum: "$items.quantity" }}},
    { $sort: { qty: -1 }},
    { $limit: 1 }
  ])

  Recent Orders:
  Order.find().populate('user items.product').sort({createdAt:-1}).limit(10)
